# MyNet Quasar 部署指南

> ## ⚠️ 先读这段：本项目只有一条部署路径
>
> **真实部署方式只有一种**：`npm run build` 产出静态文件 → rsync 到生产机的
> nginx 站点目录。入口脚本是仓根的 **`deployment.sh`**（本地 build + rsync），
> 部署目标是 `/data/app/mynet.club/my/public`，域名 `my.mynet.club`，
> 证书由主机 nginx 用 `/etc/letsencrypt/live/mynet.club/` 终结。
> 详见 `mynet_ctl/docs/knowledge-base/deployment-registry.md`（部署元数据 SSOT）。
>
> 本文档原本还夹着 **444 行「设想的备选方案」**（Docker / CDN / AWS / Vercel / Netlify /
> GitHub Actions / GitLab CI / 容器监控）——它们从未实现、照做必失败，已于 2026-09-24
> 移出归档到 [`docs/_archive/plans/部署指南-未实现的部署方案.md`](../_archive/plans/部署指南-未实现的部署方案.md)。
> **移出不是删除**：日后真要加 CDN 或容器时，从那里起手。

## 📋 部署概述

本项目**只支持一种部署方式**：静态文件部署（本地 `npm run build` → rsync 到主机 nginx）。

## 🚀 快速部署

### 方式一：静态文件部署

```bash
# 1. 构建项目
npm run build

# 2. 复制构建文件到Web服务器
cp -r dist/spa/* /var/www/mynet/

# 3. 配置Nginx
sudo cp nginx.conf /etc/nginx/sites-available/mynet
sudo ln -s /etc/nginx/sites-available/mynet /etc/nginx/sites-enabled/
sudo systemctl reload nginx
```

## 🌐 Web服务器配置

### Nginx 配置

```nginx
# /etc/nginx/sites-available/mynet
server {
    listen 80;
    listen [::]:80;
    server_name yourdomain.com www.yourdomain.com;
    
    # 重定向到HTTPS
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name yourdomain.com www.yourdomain.com;
    
    # SSL配置
    ssl_certificate /path/to/ssl/cert.pem;
    ssl_certificate_key /path/to/ssl/private.key;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-RSA-AES256-GCM-SHA512:DHE-RSA-AES256-GCM-SHA512:ECDHE-RSA-AES256-GCM-SHA384:DHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;
    
    # 根目录
    root /var/www/mynet/dist/spa;
    index index.html;
    
    # Gzip压缩
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css text/xml text/javascript application/javascript application/xml+rss application/json;
    
    # 缓存配置
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
    
    # SPA路由支持
    location / {
        try_files $uri $uri/ /index.html;
    }
    
    # API代理
    location /api/ {
        proxy_pass http://backend-api;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    
    # WebSocket代理
    location /ws/ {
        proxy_pass http://backend-api;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    
    # 安全头
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header Referrer-Policy "no-referrer-when-downgrade" always;
    add_header Content-Security-Policy "default-src 'self' http: https: data: blob: 'unsafe-inline'" always;
}

# 上游服务器
upstream backend-api {
    server 127.0.0.1:8000;
    # server 127.0.0.1:8001; # 负载均衡
}
```

### Apache 配置

```apache
<VirtualHost *:80>
    ServerName yourdomain.com
    ServerAlias www.yourdomain.com
    DocumentRoot /var/www/mynet/dist/spa
    
    # 重定向到HTTPS
    Redirect permanent / https://yourdomain.com/
</VirtualHost>

<VirtualHost *:443>
    ServerName yourdomain.com
    ServerAlias www.yourdomain.com
    DocumentRoot /var/www/mynet/dist/spa
    
    # SSL配置
    SSLEngine on
    SSLCertificateFile /path/to/ssl/cert.pem
    SSLCertificateKeyFile /path/to/ssl/private.key
    
    # 启用压缩
    LoadModule deflate_module modules/mod_deflate.so
    <Location />
        SetOutputFilter DEFLATE
        SetEnvIfNoCase Request_URI \
            \.(?:gif|jpe?g|png)$ no-gzip dont-vary
        SetEnvIfNoCase Request_URI \
            \.(?:exe|t?gz|zip|bz2|sit|rar)$ no-gzip dont-vary
    </Location>
    
    # SPA路由支持
    <Directory "/var/www/mynet/dist/spa">
        RewriteEngine On
        RewriteBase /
        RewriteRule ^index\.html$ - [L]
        RewriteCond %{REQUEST_FILENAME} !-f
        RewriteCond %{REQUEST_FILENAME} !-d
        RewriteRule . /index.html [L]
    </Directory>
    
    # 缓存配置
    <LocationMatch "\.(css|js|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$">
        ExpiresActive On
        ExpiresDefault "access plus 1 year"
    </LocationMatch>
    
    # API代理
    ProxyPass /api/ http://localhost:8000/api/
    ProxyPassReverse /api/ http://localhost:8000/api/
    
    # WebSocket代理
    ProxyPass /ws/ ws://localhost:8000/ws/
    ProxyPassReverse /ws/ ws://localhost:8000/ws/
</VirtualHost>
```



## 📊 监控和日志

### 应用监控

```typescript
// src/utils/monitoring.ts
import { createApp } from 'vue';

// 性能监控
export const setupPerformanceMonitoring = () => {
  // 页面加载时间
  window.addEventListener('load', () => {
    const loadTime = performance.timing.loadEventEnd - performance.timing.navigationStart;
    console.log('Page load time:', loadTime);
    
    // 发送到监控服务
    if (typeof gtag !== 'undefined') {
      gtag('event', 'timing_complete', {
        name: 'load',
        value: loadTime
      });
    }
  });
  
  // 错误监控
  window.addEventListener('error', (event) => {
    console.error('Global error:', event.error);
    
    // 发送错误到监控服务
    if (typeof gtag !== 'undefined') {
      gtag('event', 'exception', {
        description: event.error.message,
        fatal: false
      });
    }
  });
  
  // Vue错误处理
  const app = createApp({});
  app.config.errorHandler = (error, instance, info) => {
    console.error('Vue error:', error, info);
    
    // 发送错误到监控服务
    if (typeof gtag !== 'undefined') {
      gtag('event', 'exception', {
        description: `Vue Error: ${error.message}`,
        fatal: false
      });
    }
  };
};
```

### Nginx 日志配置

```nginx
# 自定义日志格式
log_format main_ext '$remote_addr - $remote_user [$time_local] "$request" '
                   '$status $body_bytes_sent "$http_referer" '
                   '"$http_user_agent" "$http_x_forwarded_for" '
                   '"$host" sn="$server_name" '
                   'rt=$request_time '
                   'ua="$upstream_addr" us="$upstream_status" '
                   'ut="$upstream_response_time" ul="$upstream_response_length" '
                   'cs=$upstream_cache_status';

server {
    # 访问日志
    access_log /var/log/nginx/mynet_access.log main_ext;
    
    # 错误日志
    error_log /var/log/nginx/mynet_error.log warn;
}
```

## 🔄 备份和恢复

### 自动备份脚本

```bash
#!/bin/bash
# backup.sh

DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_DIR="/backups/mynet"
APP_DIR="/var/www/mynet"

# 创建备份目录
mkdir -p $BACKUP_DIR

# 备份应用文件
tar -czf $BACKUP_DIR/mynet_app_$DATE.tar.gz -C $APP_DIR .

# 备份配置文件
tar -czf $BACKUP_DIR/mynet_config_$DATE.tar.gz /etc/nginx/sites-available/mynet

# 清理旧备份（保留7天）
find $BACKUP_DIR -name "*.tar.gz" -mtime +7 -delete

echo "Backup completed: $DATE"
```

### 恢复脚本

```bash
#!/bin/bash
# restore.sh

BACKUP_FILE=$1
APP_DIR="/var/www/mynet"

if [ -z "$BACKUP_FILE" ]; then
    echo "Usage: $0 <backup_file>"
    exit 1
fi

# 停止服务
sudo systemctl stop nginx

# 备份当前版本
mv $APP_DIR $APP_DIR.backup.$(date +%Y%m%d_%H%M%S)

# 恢复备份
mkdir -p $APP_DIR
tar -xzf $BACKUP_FILE -C $APP_DIR

# 启动服务
sudo systemctl start nginx

echo "Restore completed"
```
